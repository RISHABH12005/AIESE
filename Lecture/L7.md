# Task-1:-
Write a C++ program for a College Attendance Management System.
It should calculate the attendance percentage of each student and display the students whose attendance is below the required minimum percentage.
The program should be efficient, user-friendly, menu-driven, and easy to understand.

## AI Code:-
```C++
	#include <iostream>
	#include <vector>
	#include <iomanip>
	#include <limits>
	#include <string>

	using namespace std;

	struct Student {
	    int rollNo;
	    string name;
	    int totalClasses;
	    int attendedClasses;

	    double attendancePercentage() const {
	        if (totalClasses == 0)
	            return 0.0;

	        return (static_cast<double>(attendedClasses) / totalClasses) * 100.0;
	    }
	};

	// Clear invalid input
	void clearInput() {
	    cin.clear();
	    cin.ignore(numeric_limits<streamsize>::max(), '\n');
	}

	// Add a new student
	void addStudent(vector<Student>& students) {
	    Student s;

	    cout << "\n========== Add Student ==========\n";

	    cout << "Enter Roll Number: ";
	    while (!(cin >> s.rollNo) || s.rollNo <= 0) {
	        cout << "Invalid roll number. Enter again: ";
	        clearInput();
	    }

	    cin.ignore(numeric_limits<streamsize>::max(), '\n');

	    cout << "Enter Student Name: ";
	    getline(cin, s.name);

	    while (s.name.empty()) {
	        cout << "Name cannot be empty. Enter again: ";
	        getline(cin, s.name);
	    }

	    cout << "Enter Total Classes: ";
	    while (!(cin >> s.totalClasses) || s.totalClasses < 0) {
	        cout << "Invalid value. Enter again: ";
	        clearInput();
	    }

	    cout << "Enter Attended Classes: ";
	    while (!(cin >> s.attendedClasses) ||
	           s.attendedClasses < 0 ||
	           s.attendedClasses > s.totalClasses) {

	        cout << "Attended classes must be between 0 and "
	             << s.totalClasses << ". Enter again: ";
	        clearInput();
	    }

	    students.push_back(s);

	    cout << "\nStudent added successfully!\n";
	}

	// Display all students
	void displayStudents(const vector<Student>& students) {

	    if (students.empty()) {
	        cout << "\nNo student records available.\n";
	        return;
	    }

	    cout << "\n==================== Student Attendance ====================\n";

	    cout << left
	         << setw(10) << "Roll No"
	         << setw(25) << "Name"
	         << setw(15) << "Total"
	         << setw(15) << "Attended"
	         << setw(15) << "Attendance"
	         << "\n";

	    cout << string(80, '-') << "\n";

	    for (const auto& s : students) {

	        cout << left
	             << setw(10) << s.rollNo
	             << setw(25) << s.name
	             << setw(15) << s.totalClasses
	             << setw(15) << s.attendedClasses
	             << fixed << setprecision(2)
	             << s.attendancePercentage() << "%\n";
	    }
	}

	// Display students with low attendance
	void showLowAttendance(const vector<Student>& students,
	                       double minimumAttendance) {

	    bool found = false;

	    cout << "\n========== Low Attendance Students ==========\n";
	    cout << "Minimum Required Attendance: "
	         << minimumAttendance << "%\n\n";

	    cout << left
	         << setw(10) << "Roll No"
	         << setw(25) << "Name"
	         << setw(15) << "Attendance"
	         << "\n";

	    cout << string(50, '-') << "\n";

	    for (const auto& s : students) {

	        double percentage = s.attendancePercentage();

	        if (percentage < minimumAttendance) {

	            found = true;

	            cout << left
	                 << setw(10) << s.rollNo
	                 << setw(25) << s.name
	                 << fixed << setprecision(2)
	                 << percentage << "%\n";
	        }
	    }

	    if (!found) {
	        cout << "No students have low attendance.\n";
	    }
	}

	// Search student by roll number
	void searchStudent(const vector<Student>& students) {

	    if (students.empty()) {
	        cout << "\nNo student records available.\n";
	        return;
	    }

	    int rollNo;

	    cout << "\nEnter Roll Number: ";
	    cin >> rollNo;

	    for (const auto& s : students) {

	        if (s.rollNo == rollNo) {

	            cout << "\n========== Student Details ==========\n";
	            cout << "Roll Number : " << s.rollNo << "\n";
	            cout << "Name        : " << s.name << "\n";
	            cout << "Total Classes   : " << s.totalClasses << "\n";
	            cout << "Attended Classes: " << s.attendedClasses << "\n";
	            cout << fixed << setprecision(2);
	            cout << "Attendance  : "
	                 << s.attendancePercentage() << "%\n";

	            return;
	        }
	    }

	    cout << "\nStudent not found.\n";
	}

	// Main menu
	int main() {

	    vector<Student> students;

	    double minimumAttendance = 75.0;

	    int choice;

	    cout << "=============================================\n";
	    cout << "       COLLEGE ATTENDANCE MANAGEMENT\n";
	    cout << "=============================================\n";

	    do {

	        cout << "\n--------------- MENU ----------------\n";
	        cout << "1. Add Student\n";
	        cout << "2. Display All Students\n";
	        cout << "3. Show Low Attendance Students\n";
	        cout << "4. Search Student\n";
	        cout << "5. Change Minimum Attendance\n";
	        cout << "6. Exit\n";
	        cout << "-------------------------------------\n";

	        cout << "Enter your choice: ";

	        while (!(cin >> choice)) {
	            cout << "Invalid choice. Enter again: ";
	            clearInput();
	        }

	        switch (choice) {

	            case 1:
	                addStudent(students);
	                break;

	            case 2:
	                displayStudents(students);
	                break;

	            case 3:
	                showLowAttendance(students, minimumAttendance);
	                break;

	            case 4:
	                searchStudent(students);
	                break;

	            case 5:
	                cout << "\nEnter minimum attendance percentage: ";

	                while (!(cin >> minimumAttendance) ||
	                       minimumAttendance < 0 ||
	                       minimumAttendance > 100) {

	                    cout << "Enter a value between 0 and 100: ";
	                    clearInput();
	                }

	                cout << "Minimum attendance updated to "
	                     << minimumAttendance << "%.\n";
	                break;

	            case 6:
	                cout << "\nThank you for using the system!\n";
	                break;

	            default:
	                cout << "\nInvalid choice. Please try again.\n";
	        }

	    } while (choice != 6);

	    return 0;
	}
```

## 6 Missing details:-
- Input -> Roll number, student name, total classes, attended classes
- Attendance Calculation -> (attendedClasses / totalClasses) × 100
- Low Attendance Threshold -> Default 75%, and user can change it
- Output -> Student attendance table, low-attendance students, and student search results
- Data Storage -> vector<Student> stores records temporarily in memory
- Error Handling -> Validates numeric input, attendance range, empty names, menu choices, and threshold range
